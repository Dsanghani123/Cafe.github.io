assets/cart-drawer-ajax.js
// High-performance Vanilla JS Cart Drawer API Sync
class ShopifyCartDrawer {
  constructor() {
    this.drawer = document.querySelector('#cart-drawer-root');
    this.bindEvents();
  }

  async addItem(variantId, quantity = 1, properties = {}) {
    const response = await fetch('/cart/add.js', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ items: [{ id: variantId, quantity, properties }] })
    });
    const cartData = await response.json();
    this.updateDrawerUI(cartData);
  }
}

Developer FAQ

Technical Questions Answered

Clear answers on how I work, store migrations, and speed guarantees.

Why custom Liquid code instead of installing pre-made Shopify apps?

Pre-made apps often inject 15+ external JavaScript files that slow down your store by 2 to 4 seconds. By building features (like sticky cart drawers, custom size charts, and upsells) directly into Liquid, your store loads instantly and saves hundreds per month in app subscriptions.

Do you guarantee green Lighthouse speed scores?

Yes! For custom theme builds, I guarantee a 90+ mobile Lighthouse score and sub-1-second initial load time. I optimize image formats (WebP/AVIF), defer non-critical scripts, and eliminate unused CSS.

How do you work alongside Elena (the UI/UX Designer)?

We collaborate during the initial Figma phase so that design files match Shopify's native architecture. This eliminates re-work and ensures that every visual component is engineered cleanly from day one.
